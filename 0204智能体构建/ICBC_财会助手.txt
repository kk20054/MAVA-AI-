import gzip
import base64
import sys
import json

def gzip_compress(data):
    """GZIP压缩"""
    return gzip.compress(data.encode('utf-8'))

def gzip_decompress(compressed_data):
    """GZIP解压缩"""
    return gzip.decompress(compressed_data).decode('utf-8')

def base64_encode(data):
    """Base64编码"""
    return base64.b64encode(data).decode('ascii')

def base64_decode(encoded_data):
    """Base64解码"""
    return base64.b64decode(encoded_data.encode('ascii'))

def process(input_string):
    """处理数据：GZIP压缩 + Base64编码"""
    try:
        # 1. GZIP压缩
        compressed = gzip_compress(input_string)

        # 2. Base64编码
        encoded = base64_encode(compressed)

        return encoded
    except Exception as e:
        print(f"Error in processing: {e}")
        return None

def unprocess(encoded_string):
    """反向处理：Base64解码 + GZIP解压缩"""
    try:
        # 1. Base64解码
        decoded = base64_decode(encoded_string)

        # 2. GZIP解压缩
        decompressed = gzip_decompress(decoded)

        return decompressed
    except Exception as e:
        print(f"Error in unprocessing: {e}")
        return None

def main(args: dict) -> dict:
    # 测试数据
    test_data = "这是一个测试数据，长度超过90000个字符..." * 10  # 模拟长数据

    print(f"原始数据长度: {len(test_data)}")

    # 处理数据
    processed = process(test_data)
    print(f"处理后长度: {len(processed) if processed else 0}")

    # 解码验证
    original = args.get('query')
    restored = unprocess(original)
    print(f"还原数据长度: {len(restored) if restored else 0}")
    print(f"数据一致: {test_data == restored}")

    ret = {
        "raw_input": restored,
    }

    return ret

PERCENTAGE_NAMES = ['MA_YOY_GROWTH','MA_MOM_GROWTH','MA_L_YE_GROWTH']
INCREMENT_NAMES = ['MA_MOM_INCREMENT','MA_YOY_INCREMENT','MA_L_YE_INCREMENT']

def isIncrement(suffix_name):
    return any(increment_name in suffix_name for increment_name in INCREMENT_NAMES)

def isPercentage(suffix_name):
    return any(percentage_name in suffix_name for percentage_name in PERCENTAGE_NAMES)

# 加千分符，如果为百分比的话，乘以100，并且增加%符号。增幅增比需要有加减号
def format_amount(suffix_name,balance_str):
    try:
        amount_float = float(balance_str)
        if (isPercentage(suffix_name)):
            amount_float *= 100
        
        formatted_num = f"{amount_float:,.2f}"

        if(amount_float > 0 and (isPercentage(suffix_name) or isIncrement(suffix_name))):
            formatted_num = f"+{formatted_num}"

        if(isPercentage(suffix_name)):
            formatted_num = f"{formatted_num}%"

        return formatted_num
    except ValueError as e:
        return '输入无效'

#检查是否为月底
def check_month_end(batchDate):
    next_day = batchDate + datetime.timedelta(days=1)
    return next_day.day == 1
    
# 获取一个string 两个点之间的内容
def extract_between_points(text, start_key, end_key):
    start_marker = f"{start_key}"
    end_marker = f"{end_key}"

    start_idx = text.find(start_marker)
    if end_key is None:
        end_idx = len(text)
    else:
        end_idx = text.find(end_marker)
    print(f"{start_idx} and {end_idx}")
    if start_idx == -1 or end_idx == -1 or end_idx < start_idx:
        return ""
    return text[start_idx:end_idx]

# 将str中的时间替换为真实时间 样例 一、营业净收入-日余额-BD 改为 一、营业净收入-日余额-20251201
def replace_str_with_actual_date(orig_str,sep_symbol,output_format,date_dict):
    orig_str_split = orig_str.split(sep=sep_symbol)
    if(orig_str_split[-1] in date_dict):
        actual_date = date_dict[orig_str_split[-1]].strftime(output_format)
        orig_str = f"{''.join(orig_str_split[:-1])}{sep_symbol}{actual_date}"
    return orig_str

# 讲mapping_dict.key 转换为 mapping_dict.value。并且将实际日期替换为结果
# e.g MA_BAL-LME 转换为 日余额-20251130
def create_dict_for_ind(ind_name,mapping_dict,date_dict,input_json_obj):
    ret_dict = {}
    for input_suffix,res_suffix in mapping_dict.items():
        input_suffix = replace_str_with_actual_date(input_suffix,'-','%Y%m%d',date_dict)
        input_ind_name = f"{ind_name}-{input_suffix}"
        res_suffix = replace_str_with_actual_date(res_suffix,'_','%Y%m%d',date_dict)
        print(input_ind_name)
        if (input_ind_name in input_json_obj):
            ret_dict[res_suffix] = format_amount(input_suffix,input_json_obj[input_ind_name])
    return ret_dict

# mapping_dict定义了输出格式和输出key的内容。将输入过滤至mapping_dict定义的内容，并且映射
def extract_data_from_json(prefix_list,mapping_dict,date_dict,input_json_obj):
    res_dict = {}
    for prefix in prefix_list.keys():
        # 将指标id替换为指标中文名
        value = prefix_list[prefix]
        res_dict[value] = create_dict_for_ind(prefix,mapping_dict,date_dict,input_json_obj) 
    return res_dict

    elif index != 0 and index % 32 == 0:

        ind_dict["INDICATOR_FLAG"] = "0"

def main(args: dict) -> dict:
    """
    运行代码节点会调用此函数
    :param args: 输入固定为args字典类型，kv为输入参数键值对
    :return: 输出参数为字典类型，kv为输出参数键值对
    """
    input_str_json = json.loads(args.get('raw_input'))
    batchDate = datetime.datetime.strptime(input_str_json['batchDate'],'%Y-%m-%d').date()
    is_month_end = check_month_end(batchDate)
    monthEnd = batchDate if is_month_end else datetime.datetime.strptime(input_str_json['lastMonthLastDate'],'%Y-%m-%d').date()
    date_dict = {
        "BD":batchDate,
        "ME":monthEnd,
        "LYME":datetime.date(monthEnd.year-1,monthEnd.month,monthEnd.day),
        "LME":datetime.date(monthEnd.year,monthEnd.month,1) - datetime.timedelta(days=1),
        "LYYE":datetime.date(monthEnd.year-1,12,31)
    }
    
    year = batchDate.year
    month = batchDate.month

    # 合并两个列表为 IND_INFO
    ind_info_list = []
    
    # 处理收入类指标
    revenue_items = []
    for i, ind_name in enumerate(revenue_key_prefix_dict):
        revenue_items.append(create_json_for_indicator(ind_name, revenue_key_prefix_dict[ind_name], input_str_json, date_dict, 'revenue', '收入指标', i))
    
    # 处理资产负债类指标
    asset_liab_items = []
    for i, ind_name in enumerate(asset_liab_key_prefix_dict):
        asset_liab_items.append(create_json_for_indicator(ind_name, asset_liab_key_prefix_dict[ind_name],input_str_json, date_dict, 'asset_liab', '资产负债', len(revenue_key_prefix_dict) + i))

    # 添加rowspan信息
    page_size = 32
    revenue_length = len(revenue_items)
    asset_liab_length = len(asset_liab_items)
    total_rows = revenue_length + asset_liab_length

    for i, item in enumerate(revenue_items):
        # 计算在页面中的位置
        page_position = i % page_size
        
        if page_position == 0:
            # 页面起始行
            # 计算从当前位置开始还能有多少行
            remaining_rows = revenue_length - i
            rowspan = min(page_size, remaining_rows)
            item['ROWSPAN'] = str(rowspan)
        else:
            # 非页面起始行
            item['ROWSPAN'] = '0'

    for i, item in enumerate(asset_liab_items):
        current_index = i + revenue_length
        # 计算在页面中的位置
        page_position = current_index % page_size
        
        if i == 0 and page_position != 0:
            # 计算从当前位置开始还能有多少行
            remaining_rows = total_rows - current_index
            rowspan = min(page_size - page_position, remaining_rows)
            item['ROWSPAN'] = str(rowspan)
        elif i != 0 and page_position == 0:
            # 计算从当前位置开始还能有多少行
            remaining_rows = total_rows - current_index
            rowspan = min(page_size, remaining_rows)
            item['ROWSPAN'] = str(rowspan)
        else:
            # 非页面起始行
            item['ROWSPAN'] = '0'
        
    # 合并两个类型的数据
    ind_info_list = revenue_items + asset_liab_items

    # 计算上个月份
    last_month = month - 1 if month > 1 else 12

    oic_json_str = {
        "reportTitle":"客户经理经营诊断报告",
        "period": f"{year}年{month}月",
        "CURRENT_MONTH": month,
        "LAST_MONTH": last_month,
        "LOB_CODE":input_str_json['LOB_CODE'],
        "AC_BK_BRNO":input_str_json['AC_BK_BRNO'],
        "OIC":input_str_json['OIC'],
        "OIC_NAME":input_str_json['OIC_NAME'],
        "FINANCIAL_HIGHLIGHTS":args.get('financial_highlights'),
        "KEY_FACTS":args.get('key_facts'),
        "IND_INFO": ind_info_list
    }
    oic_final_json = json.dumps(oic_json_str, ensure_ascii = False, indent = 4, sort_keys=False)
    ret = {
        "oic_final_json": oic_final_json
    }
    return ret


import gzip
import base64
import sys
import json

def gzip_compress(data):
    """GZIP压缩"""
    return gzip.compress(data.encode('utf-8'))

def gzip_decompress(compressed_data):
    """GZIP解压缩"""
    return gzip.decompress(compressed_data).decode('utf-8')

def base64_encode(data):
    """Base64编码"""
    return base64.b64encode(data).decode('ascii')

def base64_decode(encoded_data):
    """Base64解码"""
    return base64.b64decode(encoded_data.encode('ascii'))

def process(input_string):
    """处理数据：GZIP压缩 + Base64编码"""
    try:
        # 1. GZIP压缩
        compressed = gzip_compress(input_string)

        # 2. Base64编码
        encoded = base64_encode(compressed)

        return encoded
    except Exception as e:
        print(f"Error in processing: {e}")
        return None

def unprocess(encoded_string):
    """反向处理：Base64解码 + GZIP解压缩"""
    try:
        # 1. Base64解码
        decoded = base64_decode(encoded_string)

        # 2. GZIP解压缩
        decompressed = gzip_decompress(decoded)

        return decompressed
    except Exception as e:
        print(f"Error in unprocessing: {e}")
        return None

def main(args: dict) -> dict:
    # 处理数据
    processed = process(args.get('oic_final_json'))
    ret = {
        "encrypted_res": processed,
    }

    return ret

2.Researcher Agent：信息检索与验证的核心，执行“检索-分析-精炼”循环：
-检索（Retrieve）:接收到规划智能体的结构化指令后，研究智能体首先构建初始检索式，通过接口，找到相关数据集，并以JSON对象返回（不作计算、只粘数）。
-分析（Analyze）:研究智能体会将检索数据按照“量-价-质”进行分类，并结合用户问题进行针对性回答（简单分析）。
-精炼（Refine）:在反思阶段，研究智能体会调用LLM对检索数据、分析进行相关性评估和打分（例如，从1到10分），以判断其与任务指令所要求数据是否契合，同时自主生成一个更精准的检索式，随后Agent将基于新的检索式重新进入循环，开始新一轮迭代。

3.Answerer Agent：是工作流的终点，负责将Researcher 筛选出的“金块”转化为最终交付物（分析文本+表格数据）。
