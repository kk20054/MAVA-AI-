    def parse_hierarchy(data_dict):
        """
        自动解析带有序号的中文指标名，生成层级结构 (Code, Name, Level, Parent)
        """
        sorted_items = list(data_dict.items())
        
        hierarchy_list = []
        stack = [] # 栈用于追踪父节点：[(level, code), ...]

            clean_name = name_stripped
            level = 0
            
            if re.match(r'^[一二三四五六七八九十]+、', name_stripped):

                level = 3
                clean_name = re.sub(r'^（\d+）', '', name_stripped)
            elif re.match(r'^\d+\.', name_stripped) or re.match(r'^其中：\d+\.', name_stripped):

            elif re.match(r'^[a-z]+\.', name_stripped):
                level = 5

            elif re.match(r'^[ivx]+\.', name_stripped):
                level = 6
                clean_name = re.sub(r'^[ivx]+\.', '', name_stripped)
            else:
                level = stack[-1][0] + 1 if stack else 1

            while stack and stack[-1][0] >= level:
                stack.pop()
            
            parent_code = stack[-1][1] if stack else None
            stack.append((level, code))
            
            hierarchy_list.append({
                "code": code,
                "name": clean_name.strip(),
                "level": level,
                "parent_code": parent_code
            })

你是工银亚洲MAVA智能分析助手的查询预处理模块。请严格按以下规则对用户查询进行分类与改写。

═══════════════════════════════════════
一、系统背景
═══════════════════════════════════════

本助手有三个功能模块，用户通过点击对话框下方按钮切换：
- 「查询数据」：查询工银亚洲内部OIC维度/部门维度的经营数据（详见下方「可查指标白名单」）
- 「生成报告」：一键生成OIC维度的经营分析报告
- 「知识问答」：解答财会制度、指标口径、系统操作、市场资讯等（当前激活模块）

用户当前处于「知识问答」模块。

═══════════════════════════════════════
二、「查询数据」模块可查指标白名单
═══════════════════════════════════════

以下是「查询数据」模块支持的指标范围，仅限工银亚洲本行、OIC维度或部门维度的内部经营数据：

【资产负债类】
贷款余额、贷款日均余额、存款余额、存款日均余额、管户资产余额、管户负债余额、债券投资余额、同业资产/负债

【损益类】
利息净收入、中间业务收入（手续费及佣金净收入）、汇兑损益、营业收入、营业利润、税前利润、净利润、成本收入比、资金成本、FTP相关

【资产质量类】
不良贷款余额、不良贷款率、拨备覆盖率、逾期贷款、关注类贷款

【绩效/KPI类】
RAROC、EVA、经济利润、利润分成、RWA（风险加权资产）、资本占用

【查询维度】
仅支持：本人管户（OIC）、指定EE编号的管户、部门维度
仅支持：时点余额、日均余额、月累计、年累计、同比、环比、较上年末变动

关键限制：
- 仅限工银亚洲本行数据，不包含中国工商银行（总行/母行/境内行）数据
- 不包含任何外部市场数据（利率、汇率、指数等）
- 不包含其他银行的数据

═══════════════════════════════════════
三、分类规则（按优先级从高到低）
═══════════════════════════════════════

【A. INTRO —— 问候/功能介绍】
用户在打招呼、问助手身份、问有什么功能、闲聊、问怎么使用（咨询对象为助手本身功能）。
示例："你好" "你是谁" "有什么功能" "怎么用" "hi" “你可以怎么生成/查询报告” “你可以怎么生成/查询数据”

【B. GUIDE_DATA —— 应引导至数据查询】
必须同时满足以下两个条件：
  条件1：用户想获取具体数值/金额/余额（而非概念解释）
  条件2：所问指标属于上方「可查指标白名单」中的工银亚洲内部经营数据

典型 GUIDE_DATA 示例：
- "帮我查上个月贷款余额" → 贷款余额∈白名单 ✓ 要数值 ✓ → GUIDE_DATA
- "我管户下的利息净收入是多少" → 利息净收入∈白名单 ✓ 要数值 ✓ → GUIDE_DATA
- "EE001408023的存款情况" → 存款余额∈白名单 ✓ 要数值 ✓ → GUIDE_DATA
- "看看我上月不良率" → 不良贷款率∈白名单 ✓ 要数值 ✓ → GUIDE_DATA

以下情况不是 GUIDE_DATA，即使出现了"查"字：
- 用户问的是外部市场数据（HIBOR、LIBOR、LPR、SOFR、汇率、国债收益率、股票指数等） → KNOWLEDGE
- 用户问的是中国工商银行/总行/母行/境内行的数据 → KNOWLEDGE
- 用户问的是行业整体数据（如"香港银行业贷款总量"） → KNOWLEDGE
- 用户问的是概念含义而非具体数值（如"什么是利息净收入""不良率怎么计算"） → KNOWLEDGE
- 用户问的是其他银行数据 → KNOWLEDGE

【C. GUIDE_REPORT —— 应引导至报告生成】
用户明确要求生成、导出、编写经营分析报告或诊断报告。
示例："帮我生成经营报告" "出一份上月的分析报告" "写一份业绩分析"

【D. KNOWLEDGE —— 知识问答（需检索回答）】
不属于上述A/B/C任何一类的问题，全部归入 KNOWLEDGE。包括但不限于：
- 概念/定义："什么是利息净收入" "FTP是什么意思"
- 外部市场数据："HIBOR利率" "今天美元兑港币汇率" "LPR报价"
- 行内制度/政策："差旅费报销标准" "供应商入库流程" "坏账核销审批流程"
- 操作指引："怎么在系统里提交报销" "MAVA系统怎么进行核算"
- 宏观/行业："香港金管局最新政策" "2025年香港GDP"
- 总行/母行/境内行相关："工商银行总行的净利润" "总行FTP定价政策"
- 会计准则/监管："HKFRS 9怎么规定" "HKMA对资本充足率的要求"
- 通用财会知识："坏账准备如何计提" "什么是拨备覆盖率"

- 会计准则/监管类 → 添加具体准则名称或监管机构

def main(args: dict) -> dict:
    """
    知识问答链路 - Step 2
    职责：解析 Step 1 LLM 分类结果 + 规则兜底 + 日期上下文 + 路由决策

    输入 args:
        query            – 用户原始问题 或 Step 1 LLM 输出的 JSON 字符串
        classify_result  – （可选）Step 1 LLM 输出的 JSON 字符串
        currentTime      – （可选）当前时间 YYYY-MM-DD...
    输出:
        intent           – INTRO | GUIDE_DATA | GUIDE_REPORT | KNOWLEDGE
        user_query       – 清洗后的用户原始问题
        rewritten_query  – 改写后的检索 query（仅 KNOWLEDGE 有值）
        search_keywords  – 检索关键词列表（仅 KNOWLEDGE 有值）
        is_intro         – 布尔，True 表示短路（不走检索）
        kb_query         – 知识库检索用 query
        web_query        – 联网检索用 query
        status           – SHORT_CIRCUIT | PROCEED_TO_SEARCH
        date_context     – 日期上下文 JSON 字符串
    """
    import json
    import re
    import datetime
    from datetime import timedelta

    # ════════════════════════════════════════════════════════════
    # 1. 解析输入参数
    # ════════════════════════════════════════════════════════════

    raw_query_input = str(args.get('query', '')).strip()
    classify_result = str(args.get('classify_result', '')).strip()
    current_time_str = str(args.get('currentTime', '')).strip()

    # ════════════════════════════════════════════════════════════
    # 2. 尝试从 LLM Step 1 输出中提取结构化分类
    # ════════════════════════════════════════════════════════════

    user_query = raw_query_input
    llm_intent = ''
    rewritten_query = ''
    search_keywords = []

    def try_parse_llm_json(s):
        """尝试从字符串中提取 JSON 对象"""
        if not s:
            return None
        s = s.strip()
        # 处理 markdown 代码块包裹的情况
        md_match = re.search(r'```(?:json)?\s*(\{.*?\})\s*```', s, re.DOTALL)
        if md_match:
            s = md_match.group(1)
        # 提取第一个 JSON 对象
        brace_match = re.search(r'\{[^{}]*\}', s, re.DOTALL)
        if brace_match:
            try:
                return json.loads(brace_match.group())
            except json.JSONDecodeError:
                pass
        try:
            return json.loads(s)
        except json.JSONDecodeError:
            return None

    # 优先解析 classify_result，其次尝试 raw_query_input
    parsed_llm = None
    for candidate in [classify_result, raw_query_input]:
        parsed_llm = try_parse_llm_json(candidate)
        if parsed_llm and isinstance(parsed_llm, dict) and 'intent' in parsed_llm:
            llm_intent = parsed_llm.get('intent', '').strip().upper()
            if parsed_llm.get('raw_query'):
                user_query = parsed_llm['raw_query'].strip()
            rewritten_query = parsed_llm.get('rewritten_query', '').strip()
            search_keywords = parsed_llm.get('search_keywords', [])
            break
        parsed_llm = None

    # 如果 query 是 JSON 但没 raw_query，user_query 保持原值
    # 清理 user_query 中可能残留的 JSON
    if user_query.startswith('{') and user_query.endswith('}'):
        p = try_parse_llm_json(user_query)
        if p and 'raw_query' in p:
            user_query = p['raw_query']

    # 确保 user_query 不为空
    if not user_query:
        user_query = raw_query_input

    # ════════════════════════════════════════════════════════════
    # 3. 规则引擎兜底分类（LLM 分类不可靠时纠正）
    # ════════════════════════════════════════════════════════════

    q = user_query.strip()
    q_lower = q.lower()

    # --- INTRO 规则 ---
    INTRO_PATTERNS = [
        # 纯问候
        r'^[\s]*(你好|您好|hi|hello|hey|哈喽|嗨|早上好|下午好|晚上好|早安|午安|晚安)[\s!！。.？?，,]*$',
        # 问身份
        r'(你是谁|你是什么|你叫什么|请问你是)',
        # 问功能（泛化）
        r'^.{0,4}(你能做什么|有什么功能|能帮我做什么|怎么用|如何使用|使用指引|功能介绍|使用说明)',
        # 极短无意义输入
        r'^[\s!！?？。.，,]{0,3}$',
    ]

    # --- GUIDE_DATA 规则 ---
    # 排除项：这些关键词出现时，即使句式像查数据，也不应归为 GUIDE_DATA
    EXTERNAL_MARKET_KEYWORDS = (
        r'(hibor|libor|sofr|lpr|shibor|prime rate|基准利率|同业拆息|'
        r'汇率|美元兑|港币兑|人民币兑|即期汇率|远期汇率|'
        r'国债|收益率曲线|cpi|gdp|pmi|失业率|通胀|'
        r'恒生指数|上证|道琼斯|标普|纳斯达克|股票|'
        r'总行|母行|境内行|中国工商银行|工商银行总行|'
        r'香港银行业|行业整体|全行业|市场整体|'
        r'金管局|hkma|央行|fed|美联储)'
    )

    def is_external_query(query_text):
        """判断是否涉及外部市场数据或非本行数据"""
        return bool(re.search(EXTERNAL_MARKET_KEYWORDS, query_text.lower()))

    GUIDE_DATA_PATTERNS = [
        # 明确的查询数据动词 + 工银亚洲内部财务指标名词
        r'(帮我|请|麻烦|我想|我要|能否|可以).{0,10}(查|看|查询|查看|检索|拉|调|导出).{0,20}(贷款余额|存款余额|日均余额|利息净收入|中间业务收入|手续费|营业收入|营业利润|税前利润|净利润|汇兑损益|不良贷款|不良率|拨备覆盖率|逾期贷款|关注类|RAROC|EVA|经济利润|利润分成|RWA|风险加权|资本占用|成本收入比|FTP成本|管户资产|管户负债)',
        # 我的/我管户的 + 内部指标
        r'(我的|我管户|我名下|本人).{0,15}(余额|收入|利润|损益|不良|拨备|贷款|存款|资产|负债|数据|指标|情况|表现)',
        # EE编号查数据
        r'EE\s*(?:编号)?\s*\d{6,}.{0,20}(数据|余额|收入|情况|贷款|存款|资产|负债|利润|损益|不良|拨备)',
        # 查+时间+内部指标
        r'(查|看).{0,5}(上月|上个月|本月|去年|今年|上季|同比|环比|较上年末|上半年|下半年).{0,15}(余额|收入|损益|贷款|存款|资产|负债|数据|利润|不良|拨备)',
        # 直接问"xx是多少"但指的是自己管户的具体数值
        r'(我的|管户).{0,10}(是多少|有多少|多少钱)',
    ]

    # --- GUIDE_REPORT 规则 ---
    GUIDE_REPORT_PATTERNS = [
        r'(帮我|请|麻烦|我想|我要|能否|可以).{0,10}(生成|出|写|制作|导出|做).{0,10}(报告|报表|分析报告|经营报告|诊断报告)',
        r'(经营|诊断|业绩|绩效|综合).{0,5}(报告|报表|分析)',
        r'(生成|出|做|写).{0,5}(一份|个).{0,5}(报告|分析)',
        r'(我要|我想|给我).{0,5}(报告|经营分析)',
    ]

    # 规则判定
    rule_intent = ''

    # 极短输入直接归 INTRO
    if len(q.replace(' ', '')) <= 2:
        rule_intent = 'INTRO'

    if not rule_intent:
        for p in INTRO_PATTERNS:
            if re.search(p, q_lower):
                rule_intent = 'INTRO'
                break

    if not rule_intent:
        for p in GUIDE_REPORT_PATTERNS:
            if re.search(p, q_lower):
                rule_intent = 'GUIDE_REPORT'
                break

    if not rule_intent:
        for p in GUIDE_DATA_PATTERNS:
            if re.search(p, q_lower):
                # 命中了 GUIDE_DATA 的句式，但需检查是否涉及外部数据
                if is_external_query(q_lower):
                    # 涉及外部市场/总行数据，不归 GUIDE_DATA
                    rule_intent = 'KNOWLEDGE'
                else:
                    rule_intent = 'GUIDE_DATA'
                break

    # ════════════════════════════════════════════════════════════
    # 4. 综合决策：LLM 分类 + 规则兜底
    #    规则引擎对 INTRO/GUIDE 类的判定优先级高于 LLM
    #    （因为 LLM 容易把 GUIDE 类误判为 KNOWLEDGE）
    # ════════════════════════════════════════════════════════════

    VALID_INTENTS = {'INTRO', 'GUIDE_DATA', 'GUIDE_REPORT', 'KNOWLEDGE'}

    if rule_intent in ('INTRO', 'GUIDE_DATA', 'GUIDE_REPORT'):
        # 规则命中的非知识类，优先采信规则
        final_intent = rule_intent
    elif llm_intent in VALID_INTENTS:
        final_intent = llm_intent
    elif rule_intent:
        final_intent = rule_intent
    else:
        final_intent = 'KNOWLEDGE'

    # 兼容旧版 LLM 输出 "介绍类" / "知识类"
    INTENT_ALIAS = {
        '介绍类': 'INTRO', '知识类': 'KNOWLEDGE',
        'INTRO': 'INTRO', 'GUIDE_DATA': 'GUIDE_DATA',
        'GUIDE_REPORT': 'GUIDE_REPORT', 'KNOWLEDGE': 'KNOWLEDGE',
    }
    final_intent = INTENT_ALIAS.get(final_intent, final_intent)
    if final_intent not in VALID_INTENTS:
        final_intent = 'KNOWLEDGE'

    # ════════════════════════════════════════════════════════════
    # 5. 日期上下文
    # ════════════════════════════════════════════════════════════

    if current_time_str:
        try:
            current_date = datetime.datetime.strptime(
                current_time_str[:10], "%Y-%m-%d"
            ).date()
        except ValueError:
            current_date = datetime.date.today()
    else:
        current_date = datetime.date.today()

    current_date_str = current_date.strftime("%Y-%m-%d")

    def get_month_end(year, month):
        if month == 12:
            nxt = datetime.date(year + 1, 1, 1)
        else:
            nxt = datetime.date(year, month + 1, 1)
        return nxt - timedelta(days=1)

    yesterday = current_date - timedelta(days=1)
    last_year_end = datetime.date(current_date.year - 1, 12, 31)

    month_ends = []
    ty, tm = current_date.year, current_date.month
    for _ in range(3):
        tm -= 1
        if tm <= 0:
            tm += 12
            ty -= 1
        month_ends.append(get_month_end(ty, tm))

    date_context = {
        "今天": current_date_str,
        "昨天": yesterday.strftime("%Y-%m-%d"),
        "上年末": last_year_end.strftime("%Y-%m-%d"),
        "上月末": month_ends[0].strftime("%Y-%m-%d"),
        "上上月末": month_ends[1].strftime("%Y-%m-%d"),
        "上上上月末": month_ends[2].strftime("%Y-%m-%d"),
    }

    # ════════════════════════════════════════════════════════════
    # 6. 路由输出
    # ════════════════════════════════════════════════════════════

    is_shortcircuit = final_intent in ('INTRO', 'GUIDE_DATA', 'GUIDE_REPORT')

    if is_shortcircuit:
        return {
            "intent": final_intent,
            "user_query": user_query,
            "rewritten_query": "",
            "search_keywords": json.dumps([], ensure_ascii=False),
            "is_intro": True,
            "kb_query": "",
            "web_query": "",
            "status": "SHORT_CIRCUIT",
            "date_context": json.dumps(date_context, ensure_ascii=False),
        }

    # KNOWLEDGE → 走检索
    final_rewritten = rewritten_query if rewritten_query else user_query

    return {
        "intent": final_intent,
        "user_query": user_query,
        "rewritten_query": final_rewritten,
        "search_keywords": json.dumps(
            search_keywords if search_keywords else [],
            ensure_ascii=False
        ),
        "is_intro": False,
        "kb_query": final_rewritten,
        "web_query": f"{final_rewritten}（参考日期：{current_date_str}）",
        "status": "PROCEED_TO_SEARCH",
        "date_context": json.dumps(date_context, ensure_ascii=False),
    }

import json
import ast
import math

def main(args: dict) -> dict:
    # 获取原始数据和参数
    raw_result = args.get('web_result', '')
    base_threshold = float(args.get('threshold', 0.00))
    try:
        top_k = int(args.get('top_k', 5))
    except (ValueError, TypeError):
        top_k = 5
    
    # 处理输入数据（可能是字符串化的JSON）
    if isinstance(raw_result, str):
        try:
            # 解析外层JSON
            parsed_data = json.loads(raw_result)
            # 获取真正的结果列表
            if isinstance(parsed_data, dict) and 'web_result' in parsed_data:
                results_str = parsed_data['web_result']
                # 解析内层列表结构（处理单引号问题）
                try:
                    results = json.loads(results_str)
                except json.JSONDecodeError:
                    results = ast.literal_eval(results_str)
            else:
                results = []
        except (json.JSONDecodeError, SyntaxError):
            try:
                # 尝试直接解析
                results = ast.literal_eval(raw_result)
                if isinstance(results, dict) and 'web_result' in results:
                    results = ast.literal_eval(results['web_result'])
                elif not isinstance(results, list):
                    results = []
            except (ValueError, SyntaxError):
                return {"final_result": "数据格式错误：无法解析搜索结果"}
    else:
        results = raw_result
    
    # 空结果处理
    if not results:
        return {"final_result": "未找到相关信息！"}
    
    # 预处理：提取关键字段
    processed_results = []
    for item in results:
        if not isinstance(item, dict):
            continue
            
        # 提取relevance（处理字符串/空值）
        relevance = item.get("relevance")
        if relevance is None:
            relevance = 0.0
        elif not isinstance(relevance, float):
            try:
                relevance = float(relevance)
            except (ValueError, TypeError):
                relevance = 0.0
                
        # 提取content
        content = item.get("content", "")
        if not isinstance(content, str):
            content = ""
        
        # 添加到处理后的列表
        processed_results.append({
            "relevance": relevance,
            "content": content[:4000],  # 预截断
            "raw_data": item  # 保留原始数据
        })
    
    # === 动态阈值筛选逻辑 ===
    # 步骤1: 基础过滤 (移除低于基础阈值的结果)
    filtered = [
        item for item in processed_results
        if item["relevance"] >= base_threshold
    ]
    
    # 无基础过滤结果时返回原始最高分Top-K
    if not filtered:
        sorted_results = sorted(
            processed_results,
            key=lambda x: x["relevance"],
            reverse=True
        )[:min(top_k, len(processed_results))]
        return {"final_result": json.dumps(sorted_results, ensure_ascii=False)}
    
    # 步骤2: 动态阈值计算
    scores = [item["relevance"] for item in filtered]
    sorted_scores = sorted(scores)
    n = len(sorted_scores)
    
    # 计算50%分位数
    p75_index = 0.5 * (n - 1)
    floor_idx = math.floor(p75_index)
    ceil_idx = math.ceil(p75_index)
    
    if floor_idx == ceil_idx:
        p75 = sorted_scores[floor_idx]
    else:
        weight = p75_index - floor_idx
        p75 = (1 - weight) * sorted_scores[floor_idx] + weight * sorted_scores[ceil_idx]
    
    # 计算动态阈值
    max_score = max(scores)
    dynamic_thresh = max(p75, max_score * 0.8)
    
    # 步骤3: 动态阈值筛选
    high_rel = [
        item for item in filtered
        if item["relevance"] >= dynamic_thresh
    ]
    
    # 步骤4: 保底机制
    if not high_rel:
        filtered_sorted = sorted(
            filtered,
            key=lambda x: x["relevance"],
            reverse=True
        )
        high_rel = filtered_sorted[:min(top_k, len(filtered_sorted))]
    
    # 步骤5: 排序并返回Top-K
    high_rel_sorted = sorted(
        high_rel,
        key=lambda x: x["relevance"],
        reverse=True
    )[:top_k]
    
    return {"final_result": json.dumps(high_rel_sorted, ensure_ascii=False)}

def main(args: dict) -> dict:
    """
    知识问答链路 - Step 3.1

        top_k = int(args.get('top_k', 3))

        top_k = 3

    # ── 解析输入 ──
    results = []
    if isinstance(raw_result, list):
        results = raw_result
    elif isinstance(raw_result, dict):
        results = raw_result.get("result", raw_result.get("results", []))
    elif isinstance(raw_result, str):
        try:
            parsed = json.loads(raw_result)
            if isinstance(parsed, list):
                results = parsed
            elif isinstance(parsed, dict):
                results = parsed.get("result", parsed.get("results", []))
        except json.JSONDecodeError:
            try:
                parsed = ast.literal_eval(raw_result)
                if isinstance(parsed, dict):
                    results = parsed.get("result", [])
                elif isinstance(parsed, list):
                    results = parsed
            except (ValueError, SyntaxError):
                return {"final_result": json.dumps([], ensure_ascii=False), "has_result": False}

    if not results:
        return {"final_result": json.dumps([], ensure_ascii=False), "has_result": False}

    processed = []

        if not isinstance(data, dict):
            continue

        relevance = data.get("relevance", data.get("score", 0.0))
        try:
            relevance = float(relevance)
        except (ValueError, TypeError):
            relevance = 0.0

        snippet = data.get("snippet", {})
        if isinstance(snippet, dict):
            content = str(snippet.get("content", ""))
        elif isinstance(data.get("content"), str):
            content = data["content"]

        if not content.strip():

        processed.append({
            "relevance": relevance,
            "content": content[:6000],

    if not processed:
        return {"final_result": json.dumps([], ensure_ascii=False), "has_result": False}

    filtered = [x for x in processed if x["relevance"] >= base_threshold]

        # 低于基础阈值：取 top_k 但标记低置信
        fallback = sorted(processed, key=lambda x: x["relevance"], reverse=True)[:top_k]
        for item in fallback:
            item["low_confidence"] = True
        return {
            "final_result": json.dumps(fallback, ensure_ascii=False),
            "has_result": len(fallback) > 0,
        }

    # 第二轮：动态阈值（P75 与 max*0.8 取较大值）
    scores = sorted([x["relevance"] for x in filtered])
    n = len(scores)
    p75_idx = 0.75 * (n - 1)
    fl, cl = math.floor(p75_idx), math.ceil(p75_idx)
    p75 = scores[fl] if fl == cl else scores[fl] * (1 - (p75_idx - fl)) + scores[cl] * (p75_idx - fl)
    dynamic_thresh = max(p75, max(scores) * 0.8)

    high_rel = [x for x in filtered if x["relevance"] >= dynamic_thresh]
    if not high_rel:
        high_rel = sorted(filtered, key=lambda x: x["relevance"], reverse=True)[:top_k]

    final = sorted(high_rel, key=lambda x: x["relevance"], reverse=True)[:top_k]

    return {
        "final_result": json.dumps(final, ensure_ascii=False),
        "has_result": len(final) > 0,
    }

    def parse_search_results(raw, label):
        """将检索结果 JSON 转为可读文本块"""
        if not raw or str(raw).strip() in ('', '[]', '{}'):
            return f'（{label}无匹配结果）'
        try:
            items = json.loads(raw) if isinstance(raw, str) else raw
        except (json.JSONDecodeError, TypeError):
            return f'（{label}解析异常）'

            return f'（{label}无匹配结果）'

        blocks = []
        for i, item in enumerate(items, 1):
            if not isinstance(item, dict):
                continue
            content   = item.get('content', '').strip()
            relevance = item.get('relevance', '')
            title     = item.get('title', '')
            low_conf  = item.get('low_confidence', False)

                continue

            if title:
                header += f' {title}'
            if relevance:
                header += f'（相关度: {relevance}）'
            if low_conf:
                header += '（低置信度）'
            blocks.append(f'{header}\n{content}')

        return '\n\n'.join(blocks) if blocks else f'（{label}无有效内容）'
