import json

def main(args: dict) -> dict:
    json_str = args.get('data').replace("'{", '{').replace("}'", '}')
    result = []

    try:
        data = json.loads(json_str)
        # 新增：在解析后直接尝试提取 amount 值
        extracted_qryamt = ''
        if isinstance(data, dict) and 'amount' in data:
            amount_list = data['amount']
            if isinstance(amount_list, list) and len(amount_list) > 0:
                first_amount_item = amount_list[0]
                if isinstance(first_amount_item, dict) and 'value' in first_amount_item:
                    extracted_qryamt = first_amount_item['value']

        # 原有平铺数据结构逻辑保持不变
        if isinstance(data, dict):
            for key, value in data.items():
                prefix = ""
                new_key = f"{prefix}.{key}" if prefix else key
                if isinstance(value, (dict, list)):
                    nested_result = []
                    if isinstance(value, dict):
                        for k, v in value.items():
                            nested_prefix = f"{new_key}.{k}"
                            if isinstance(v, (dict, list)):
                                nested_result.append((nested_prefix, str(v)))
                            else:
                                nested_result.append((nested_prefix, v))
                    elif isinstance(value, list):
                        for i, item in enumerate(value):
                            nested_prefix = f"{new_key}[{i}]"
                            if isinstance(item, (dict, list)):
                                nested_result.append((nested_prefix, str(item)))
                            else:
                                nested_result.append((nested_prefix, item))
                    result.extend(nested_result)
                else:
                    result.append((new_key, value))
        elif isinstance(data, list):
            for i, item in enumerate(data):
                prefix = ""
                new_key = f"{prefix}[{i}]"
                if isinstance(item, (dict, list)):
                    nested_result = []
                    if isinstance(item, dict):
                        for k, v in item.items():
                            nested_prefix = f"{new_key}.{k}"
                            if isinstance(v, (dict, list)):
                                nested_result.append((nested_prefix, str(v)))
                            else:
                                nested_result.append((nested_prefix, v))
                    elif isinstance(item, list):
                        for j, sub_item in enumerate(item):
                            nested_prefix = f"{new_key}[{j}]"
                            if isinstance(sub_item, (dict, list)):
                                nested_result.append((nested_prefix, str(sub_item)))
                            else:
                                nested_result.append((nested_prefix, sub_item))
                    result.extend(nested_result)
                else:
                    result.append((new_key, item))
        else:
            result.append(("", data))

        # 提取其他字段
        extracted_start_date = ''
        extracted_end_date = ''
        customer_type = ''

        for item in result:
            if isinstance(item, tuple) and len(item) >= 2:
                key, value = item[0], item[1]
                if key == 'start_date':
                    extracted_start_date = value
                elif key == 'end_date':
                    extracted_end_date = value
                elif key == 'customer_type':
                    if value == '行外客户':
                        customer_type = '1'
                    elif value == '行内客户':
                        customer_type = '2'
                # 新增: 从平铺结果中再次确认 amount 值 (双重保险)
                elif key == 'amount[0].value' and not extracted_qryamt:
                    extracted_qryamt = value

        return {
            "start_date": extracted_start_date,
            "end_date": extracted_end_date,
            "customer_type": customer_type,
            "qryamt": extracted_qryamt  # 这里返回提取的数值
        }

    except Exception as e:
        return {"error": str(e)}
